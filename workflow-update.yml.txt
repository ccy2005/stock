name: 每日更新股票資料

on:
  schedule:
    # GitHub 用 UTC。台灣時間 = UTC + 8
    - cron: "40 10 * * 1-5"   # 台灣 18:40：日K + 三大法人
    - cron: "50 13 * * 1-5"   # 台灣 21:50：補抓較晚公布的資料
  workflow_dispatch:          # 可在 Actions 頁面手動按「Run workflow」

permissions:
  contents: write

concurrency:
  group: update-data
  cancel-in-progress: false

jobs:
  update:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: 抓資料
        env:
          FINMIND_TOKEN: ${{ secrets.FINMIND_TOKEN }}
        run: python update.py
      - name: 有變動才提交
        run: |
          git config user.name "stock-bot"
          git config user.email "stock-bot@users.noreply.github.com"
          git add data
          git diff --cached --quiet && echo "沒有變動" || (git commit -m "data: $(TZ=Asia/Taipei date +'%F %H:%M')" && git push)
